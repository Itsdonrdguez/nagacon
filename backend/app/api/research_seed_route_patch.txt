from fastapi import Query

@router.post("/opportunities/{opportunity_id}/seed-leads")
def seed_leads(
    opportunity_id: int,
    seed_mode: str = Query("product_only", pattern="^(all|product_only)$"),
    db: Session = Depends(get_db),
):
    opp = db.query(Opportunity).filter(Opportunity.id == opportunity_id).first()
    if not opp:
        raise HTTPException(status_code=404, detail="Opportunity not found")
    research = search_usaspending_for_opportunity(opp)
    return seed_usaspending_vendors_into_leads(db, opp, research, seed_mode=seed_mode)
